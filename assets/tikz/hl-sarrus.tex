% Huskelapp: Sarrus regel for 3x3-determinant.
% alt: Matrisen med de to første kolonnene skrevet en gang til. Tre diagonaler ned mot høyre gir pluss, tre opp mot høyre gir minus
\documentclass[tikz,border=2pt]{standalone}
\input{krets}
\begin{document}
\begin{tikzpicture}
\matrix (m) [matrix of math nodes, nodes={minimum width=0.6cm, minimum height=0.5cm, inner sep=1pt}] {
  a & b & c & a & b \\ d & e & f & d & e \\ g & h & i & g & h \\ };
\draw[mut] (m-1-3.north east) -- (m-3-3.south east);
\foreach \c in {1,2,3} { \pgfmathtruncatemacro{\d}{\c+2}
  \draw[acc, -{Stealth[length=3pt]}, shorten <=1pt] (m-1-\c.north west) -- (m-3-\d.south east); }
\foreach \c in {1,2,3} { \pgfmathtruncatemacro{\d}{\c+2}
  \draw[hi, -{Stealth[length=3pt]}, dashed, shorten <=1pt] (m-3-\c.south west) -- (m-1-\d.north east); }
\node[acc, right] at (m-1-5.east) {$+$}; \node[hi, right] at (m-3-5.east) {$-$};
\end{tikzpicture}
\end{document}
